#!/usr/bin/env python3
"""Release the backend-owned E-stop via the guarded, authenticated API,
exactly as documented in docs/PX4_NSH_OVER_SSH.md. Credentials and token
are held only in this process's memory and are never printed."""
import json, urllib.request, urllib.error

from rover_backend.config import settings  # relies on sourced ROS + workspace setup.bash

BASE = 'http://127.0.0.1:5001'


def call(method, path, token=None, body=None):
    data = json.dumps(body).encode() if body is not None else None
    req = urllib.request.Request(BASE + path, data=data, method=method)
    req.add_header('Content-Type', 'application/json')
    if token:
        req.add_header('Authorization', 'Bearer ' + token)
    try:
        with urllib.request.urlopen(req, timeout=10) as resp:
            return resp.status, json.loads(resp.read().decode())
    except urllib.error.HTTPError as e:
        return e.code, json.loads(e.read().decode())


def main():
    status, login = call('POST', '/api/auth/login',
                          body={'username': settings.static_username, 'password': settings.static_password})
    if status != 200 or not login.get('success'):
        print('LOGIN_FAILED', status, json.dumps({k: v for k, v in login.items() if k != 'token'}))
        return
    token = login['token']
    print('LOGIN_OK session_id=' + login.get('session_id', '?'))

    status, release = call('POST', '/api/estop/release', token=token, body={})
    print('RELEASE_STATUS', status)
    print('RELEASE_BODY', json.dumps(release))

    status, logout = call('POST', '/api/auth/logout', token=token)
    print('LOGOUT_STATUS', status)


if __name__ == '__main__':
    main()
