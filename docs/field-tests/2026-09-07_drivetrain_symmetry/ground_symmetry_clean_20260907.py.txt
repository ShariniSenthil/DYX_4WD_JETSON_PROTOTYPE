#!/usr/bin/env python3
import os,sys,time,threading,signal,re,json
from pathlib import Path
os.environ['MAVLINK20']='1';sys.path.insert(0,'/tmp/ground_symmetry_deps')
import rclpy
from rclpy.executors import SingleThreadedExecutor
from rclpy.signals import SignalHandlerOptions
from rclpy.qos import QoSProfile,ReliabilityPolicy
from mavros_msgs.msg import Mavlink,StatusText
from mavros_msgs.srv import CommandBool,SetMode
from mavros.mavlink import convert_to_rosmsg,convert_to_bytes
from pymavlink.dialects.v20 import common as mav
from ground_symmetry_20260907 import Test,SEQUENCE

class CleanTest(Test):
 def __init__(self):
  self.streaming=False;self.monitor_guard=False;self.manual_guard=False
  self.mc_time=0.;self.mc=None;self.mc_samples=0;self.mc_stamp=0;self.mc_block=''
  self.shell_text='';self.ulog=None;self.failsafe_messages=[];self.current_y=0;self.changed_at=0
  self.neutral_check=False
  super().__init__()
  self.encoder=mav.MAVLink(None,srcSystem=1,srcComponent=191);self.decoder=mav.MAVLink(None)
  q=QoSProfile(depth=1000,reliability=ReliabilityPolicy.BEST_EFFORT)
  self.rawpub=self.create_publisher(Mavlink,'/uas1/mavlink_sink',10)
  self.create_subscription(Mavlink,'/uas1/mavlink_source',self.raw_rx,q)
  self.create_subscription(StatusText,'/mavros/statustext/recv',self.status_rx,q)
  self.shell_file=open('/tmp/ground_symmetry_clean_px4_monitor.txt','w',buffering=1)
 def publish(self,y):
  if y!=self.current_y:self.changed_at=time.monotonic();self.current_y=y
  super().publish(y)
 def tick(self):
  if not self.streaming:return
  if self.monitor_guard and not self.abort_reason:
   if time.monotonic()-self.mc_time>.65:self.abort('PX4 manual_control_setpoint monitor stale')
  if self.neutral_check and self.state and self.state.armed and self.current_y==0 and time.monotonic()-self.changed_at>.25:
   if self.rc is None or time.monotonic()-self.rc_time>.65:self.abort('PWM telemetry stale during armed neutral')
   elif len(self.rc.channels)<4 or any(abs(v-1500)>1 for v in self.rc.channels[:4]):
    self.abort('Non-neutral PWM while neutral requested: '+str(self.rc.channels[:4]))
  super().tick()
 def on_state(self,m):
  super().on_state(m)
  if self.manual_guard and (m.mode!='MANUAL' or m.system_status in (5,6)):
   self.abort('MANUAL lost or FCU critical status: '+m.mode+'/'+str(m.system_status))
 def status_rx(self,m):
  match=re.search(r'(/fs/microsd/log/\S+\.ulg)',m.text)
  if match:self.ulog=match.group(1)
  lower=m.text.lower()
  if 'failsafe' in lower and any(x in lower for x in ('activated','triggered','enabled')):
   self.failsafe_messages.append(m.text)
   if self.manual_guard:self.abort('PX4 '+m.text)
 def raw_rx(self,m):
  if m.msgid!=126:return
  try: packets=self.decoder.parse_buffer(convert_to_bytes(m)) or []
  except Exception as e:
   if self.monitor_guard:self.abort('PX4 shell decoding failed: '+str(e))
   return
  for packet in packets:
   if packet.get_type()!='SERIAL_CONTROL':continue
   text=bytes(packet.data[:packet.count]).decode(errors='replace')
   self.shell_file.write(text);self.shell_text+=text
   self.mc_block+=text
   # A complete topic print ends with sticks_moving. Parse only complete records.
   while 'sticks_moving:' in self.mc_block and '\n' in self.mc_block.split('sticks_moving:',1)[1]:
    idx=self.mc_block.index('\n',self.mc_block.index('sticks_moving:'))
    block=self.mc_block[:idx+1];self.mc_block=self.mc_block[idx+1:]
    begin=block.rfind('TOPIC: manual_control_setpoint')
    if begin<0:continue
    block=block[begin:]
    def field(name):
     match=re.search(r'^\s*'+name+r':\s*([^\r\n]+)',block,re.M)
     return match.group(1).strip() if match else None
    try:
     stamp=int(field('timestamp').split()[0]);valid=field('valid')=='True'
     source=int(field('data_source'));throttle=float(field('throttle'))
     age_match=re.search(r'timestamp:\s*\d+\s*\(([\d.]+) seconds ago\)',block)
     age=float(age_match.group(1)) if age_match else 0.
     sample=int(field('timestamp_sample').split()[0]);lag=(stamp-sample)/1e6
     data=dict(valid=valid,source=source,throttle=throttle,roll=float(field('roll')),timestamp=stamp,age=age,sample_lag=lag)
     fresh=valid and 2<=source<=7 and stamp>self.mc_stamp and age<.20 and lag<.20 and throttle==0.
     if fresh:self.mc_samples+=1
     self.mc=data;self.mc_stamp=stamp;self.mc_time=time.monotonic()
     if self.monitor_guard and not fresh:self.abort('PX4 manual_control_setpoint check failed: '+str(data))
    except (TypeError,ValueError,AttributeError) as e:
     if self.monitor_guard:self.abort('Incomplete PX4 setpoint record: '+str(e))
   if len(self.mc_block)>12000:self.mc_block=self.mc_block[-6000:]
 def send_shell(self,text,flags=mav.SERIAL_CONTROL_FLAG_RESPOND):
  raw=text.encode()
  chunks=[raw[i:i+70] for i in range(0,len(raw),70)] or [b'']
  for chunk in chunks:
   msg=self.encoder.serial_control_encode(mav.SERIAL_CONTROL_DEV_SHELL,flags,0,0,len(chunk),list(chunk)+[0]*(70-len(chunk)))
   msg.pack(self.encoder);self.encoder.seq=(self.encoder.seq+1)%256;self.rawpub.publish(convert_to_rosmsg(msg))
 def shell(self,command,timeout=8):
  offset=len(self.shell_text);self.send_shell(command+'\n');end=time.monotonic()+timeout
  while time.monotonic()<end:
   result=self.shell_text[offset:]
   if command in result and 'nsh>' in result[result.find(command)+len(command):]:return re.sub(r'\x1b\[[0-9;]*[A-Za-z]','',result)
   time.sleep(.02)
  raise RuntimeError('PX4 shell command timeout: '+command)
 def end_monitor(self):
  self.monitor_guard=False;self.manual_guard=False;self.send_shell('\x03')
  time.sleep(.4)

def main():
 rclpy.init(signal_handler_options=SignalHandlerOptions.NO)
 n=CleanTest();ex=SingleThreadedExecutor();ex.add_node(n);thread=threading.Thread(target=ex.spin,daemon=True);thread.start()
 for sig in (signal.SIGINT,signal.SIGTERM,signal.SIGHUP):signal.signal(sig,lambda number,frame:n.abort('Signal '+str(number)))
 report={'manual_entered':False,'armed_neutral_stable':False,'restored':False,'errors':[]}
 try:
  n.wait_for(lambda:n.state is not None and n.estop is not None,10,'state telemetry')
  if n.state.armed:
   q=CommandBool.Request();q.value=False
   if not n.call('arm',q).success:raise RuntimeError('Disarm rejected')
   n.wait_for(lambda:not n.state.armed,5,'disarmed confirmation')
  print('STEP_1_DISARMED_CONFIRMED',flush=True)
  n.rc_config=n.params(['RC_MAP_ARM_SW','RC_ARMSWITCH_TH'],fresh=True)
  channel=int(n.rc_config['RC_MAP_ARM_SW'])
  if channel:n.rc_config.update(n.params(['RC'+str(channel)+'_'+k for k in ['MIN','MAX','TRIM','DZ','REV']],fresh=False))
  n.wait_for(lambda:n.rawpub.get_subscription_count()>0,5,'PX4 shell route')
  print('SHELL_READY',n.shell('echo symmetry_clean_ready'),flush=True)
  print(n.shell('param set COM_RC_IN_MODE 1'),flush=True)
  saved=n.shell('param save');print(saved,flush=True)
  shown=n.shell('param show COM_RC_IN_MODE');print(shown,flush=True)
  if not re.search(r'COM_RC_IN_MODE[^\r\n:]*:\s*1\b',shown):raise RuntimeError('PX4 did not verify COM_RC_IN_MODE=1')
  if not n.state or n.state.armed:raise RuntimeError('Rover armed before neutral startup')
  print('LOGGER_BEFORE',n.shell('logger status'),flush=True)
  print('PX4_STATUS_BEFORE',n.shell('listener vehicle_status -n 1'),flush=True)
  if n.count_publishers('/mavros/manual_control/send')!=1:raise RuntimeError('Competing manual-control publisher')
  n.streaming=True;print('STEP_3_NEUTRAL_STREAM_STARTED_50HZ',flush=True)
  n.send_shell('listener manual_control_setpoint -n 100000 -r 5\n')
  n.wait_for(lambda:n.mc_samples>=10,10,'ten fresh valid PX4 MAVLink manual-control samples')
  print('STEP_4_PX4_SETPOINT_VERIFIED',n.mc,flush=True);n.monitor_guard=True
  if n.estop is not False:raise RuntimeError('Backend E-stop is latched')
  if n.state.armed:raise RuntimeError('Operator armed before MANUAL stability verification')
  q=SetMode.Request();q.custom_mode='MANUAL'
  if not n.call('mode',q).mode_sent:raise RuntimeError('MANUAL request failed')
  n.wait_for(lambda:n.state.mode=='MANUAL' and not n.state.armed,5,'disarmed MANUAL confirmation')
  n.manual_guard=True;start=time.monotonic()
  n.wait_for(lambda:time.monotonic()-start>=3,4,'three seconds stable disarmed MANUAL')
  if n.state.armed:raise RuntimeError('Armed before three seconds of disarmed MANUAL stability')
  report['manual_entered']=True
  print('STEP_6_MANUAL_STABLE_3_SECONDS; READY_FOR_OPERATOR_MANUAL_ARM',flush=True)
  n.wait_for(lambda:n.state.armed,600,'manual arming by operator')
  n.active=True;n.neutral_check=True;n.changed_at=time.monotonic()
  n.wait_for(lambda:n.ulog is not None,3,'PX4 SD ULog start announcement')
  print('ULOG',n.ulog,flush=True)
  with n.lock:n.start=time.monotonic()
  n.wait_for(lambda:n.step>=1 or n.done.is_set(),4,'three seconds armed neutral')
  if n.abort_reason:raise RuntimeError(n.abort_reason)
  report['armed_neutral_stable']=True
  n.wait_for(lambda:n.done.is_set(),42,'remaining sequence')
  if n.abort_reason:raise RuntimeError(n.abort_reason)
 except Exception as e:
  report['errors'].append(str(e));n.abort(str(e))
 finally:
  n.monitor_guard=False;n.manual_guard=False;n.active=False;n.neutral_check=False
  with n.lock:n.start=None;n.streaming=True;n.publish(0)
  time.sleep(3);n.end_monitor()
  try:
   if n.state and n.state.armed:
    q=CommandBool.Request();q.value=False
    if not n.call('arm',q).success:raise RuntimeError('Cleanup disarm rejected')
    n.wait_for(lambda:not n.state.armed,8,'cleanup disarmed',allow_abort=True)
   print('STEP_11_DISARMED',flush=True);time.sleep(3)
   logger=n.shell('logger status');report['logger_after']=logger;print('LOGGER_AFTER',logger,flush=True)
   if n.rc_config:
    print('WAIT_RC_ARM_SWITCH_DISARM',flush=True)
    n.wait_for(n.arm_switch_off,600,'RC switch disarm before restoring input',allow_abort=True)
   print(n.shell('param set COM_RC_IN_MODE 0'),flush=True)
   save=n.shell('param save');print(save,flush=True);report['param_save_output']=save
   shown=n.shell('param show COM_RC_IN_MODE');print(shown,flush=True)
   report['restored']=bool(re.search(r'COM_RC_IN_MODE[^\r\n:]*:\s*0\b',shown));report['param_show_output']=shown
   if not report['restored']:raise RuntimeError('Restore verification failed')
   time.sleep(2)
   if n.state.armed:
    q=CommandBool.Request();q.value=False;n.call('arm',q)
    n.wait_for(lambda:not n.state.armed,5,'post-restoration disarm',allow_abort=True)
    report['errors'].append('RC switch rearmed after restoration; disarmed again')
  except Exception as e:report['errors'].append('CLEANUP: '+str(e));print('CLEANUP_ERROR',e,flush=True)
  report.update(sequence_executed=n.executed,ulog=n.ulog,abort_reason=n.abort_reason,failsafe_messages=n.failsafe_messages,last_px4_setpoint=n.mc)
  Path('/tmp/ground_symmetry_clean_20260907_result.json').write_text(json.dumps(report,indent=2));print('RESULT',json.dumps(report),flush=True)
  time.sleep(3);n.send_shell('',flags=0);time.sleep(.2)
  ex.shutdown();thread.join(timeout=3);n.trace.close();n.shell_file.close();n.destroy_node();rclpy.shutdown()
if __name__=='__main__':main()
